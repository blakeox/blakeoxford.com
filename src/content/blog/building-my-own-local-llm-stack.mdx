<ContentBand
  align="center"
  kicker="Implementation Journey"
  title="Building It Out"
  description="Five steps from concept to production"
  class="my-16"
/>

<div class="not-prose">
  <Timeline>
    <TimelineItem step={1} variant="accent" title="Making Ollama a Managed Service">
      <p class="mb-6 text-lg leading-relaxed opacity-90 md:text-xl">
        Ollama originally ran as a background process under my admin account. I refactored it into a{' '}
        <strong>LaunchDaemon</strong> under a non-admin user (
        <code class="rounded bg-foreground/10 px-3 py-1 font-mono text-base">"ollama"</code>),
        isolating model execution and file access.
      </p>
      <Callout variant="accent" title="Result">
        <p class="text-base opacity-85">
          Even if something goes sideways, it stays contained and restarts automatically.
        </p>
      </Callout>
    </TimelineItem>

    <TimelineItem step={2} variant="primary" title="Hardening SSH">
      <p class="mb-6 text-lg leading-relaxed opacity-90 md:text-xl">
        Default macOS SSH is friendly — <strong>too friendly</strong>. I locked it down to:
      </p>
      <div class="my-8 grid gap-4 sm:grid-cols-3">
        <FeatureCard variant="primary" title="Key-based auth only" hover={false} />
        <FeatureCard variant="primary" title="Root disabled" hover={false} />
        <FeatureCard variant="primary" title="Single whitelisted user" hover={false} />
      </div>
      <Callout variant="info" title="Result">
        <p class="text-base opacity-85">Still convenient, but hardened to a professional baseline.</p>
      </Callout>
    </TimelineItem>

    <TimelineItem step={3} variant="accent" title="Running WebUI via Docker">
      <p class="mb-6 text-lg leading-relaxed opacity-90 md:text-xl">
        Open WebUI runs in a container bound only to{' '}
        <code class="rounded bg-foreground/10 px-3 py-1 font-mono text-base">localhost</code> for
        safety and persistence.
      </p>
      <Callout variant="accent" class="my-6">
        <p class="text-lg italic leading-relaxed">
          Docker keeps the interface isolated, data stored locally, and restarts handled
          automatically with{' '}
          <code class="rounded bg-foreground/10 px-3 py-1 font-mono text-sm not-italic">
            --restart unless-stopped
          </code>
          .
        </p>
      </Callout>
      <Callout variant="accent" title="Result">
        <p class="text-base opacity-85">
          Private interface with vector memory persistence and data portability.
        </p>
      </Callout>
    </TimelineItem>

    <TimelineItem step={4} variant="primary" title="Adding Self-Healing Automation">
      <p class="mb-6 text-lg leading-relaxed opacity-90 md:text-xl">
        I wrote a lightweight <strong>watchdog script</strong> that checks container health and
        restarts it if it ever stops responding.
      </p>
      <Callout variant="info" title="Result">
        <p class="text-base opacity-85">
          Automatic failover loop — quiet reliability with no manual babysitting.
        </p>
      </Callout>
    </TimelineItem>

    <TimelineItem step={5} variant="accent" title="Seamless Boot Recovery">
      <p class="mb-6 text-lg leading-relaxed opacity-90 md:text-xl">
        LaunchDaemons handle startup sequencing — <strong>Docker → Ollama → WebUI</strong>.
      </p>
      <Callout variant="accent" title="Result">
        <p class="text-base opacity-85">
          If power cuts or macOS updates, the entire system rebuilds itself without input.
        </p>
      </Callout>
    </TimelineItem>

  </Timeline>
</div>
